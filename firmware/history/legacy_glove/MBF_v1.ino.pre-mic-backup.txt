#include <Wire.h>
#include <Adafruit_MPU6050.h>
#include <Adafruit_Sensor.h>

#include <WiFi.h>
#include <PubSubClient.h>
#include <esp_timer.h>

// ============================================================
// USER SETTINGS
// ============================================================

const char WIFI_SSID[]     = "HUAWEI-cU9R";
const char WIFI_PASSWORD[] = "CHANGE_ME_PRIVATE";

// IPv4 address of the laptop running Mosquitto.
// Example: "192.168.1.105"
const char MQTT_BROKER_IP[] = "192.168.101.16";

const uint16_t MQTT_BROKER_PORT = 1883;

#define DEVICE_ID "glove"

const char MOTION_TOPIC[]       = "sensor/glove/gesture";
const char BATTERY_TOPIC[]      = "status/" DEVICE_ID "/battery";
const char CONNECTION_TOPIC[]   = "status/" DEVICE_ID "/connection";

const char CONNECTION_ONLINE_PAYLOAD[] =
    "{\"schema_version\":1,\"type\":\"connection_status\","
    "\"source_id\":\"" DEVICE_ID "\",\"state\":\"ONLINE\"}";

const char CONNECTION_OFFLINE_PAYLOAD[] =
    "{\"schema_version\":1,\"type\":\"connection_status\","
    "\"source_id\":\"" DEVICE_ID "\",\"state\":\"OFFLINE\"}";

// Development connection without TLS.
// Use only on your private router or phone hotspot.

// ============================================================
// HARDWARE
// ============================================================

Adafruit_MPU6050 mpu;

const uint8_t MPU_ADDRESS = 0x68;

const int FLEX_THUMB_PIN  = 39;  // A1
const int FLEX_INDEX_PIN  = 34;  // A2
const int FLEX_MIDDLE_PIN = 35;  // A3

// Temporary values. Calibrate each threshold using the raw Serial readings.
// With the current divider wiring, the ADC reading falls as the sensor bends.
const uint16_t FLEX_THUMB_BENT_THRESHOLD  = 2000;
const uint16_t FLEX_INDEX_BENT_THRESHOLD  = 2000;
const uint16_t FLEX_MIDDLE_BENT_THRESHOLD = 2000;

// DFR0478 A0 corresponds to ESP32 GPIO36.
// V4 schematic shows an onboard 1:1 battery divider.
const int BATTERY_ADC_PIN = 36;  // A0

// Equal-value divider: ADC receives approximately VBAT / 2.
const float BATTERY_DIVIDER_RATIO = 2.0f;

// Adjust after comparison with a multimeter.
// Example: multimeter=3.92 V, firmware=3.80 V
// calibration = 3.92 / 3.80 = 1.0316
const float BATTERY_CALIBRATION = 0.988f;

const uint32_t SAMPLE_PERIOD_MS       = 20;  // 50 Hz
const uint32_t BATTERY_PERIOD_MS      = 1000;
const uint32_t BATTERY_PUBLISH_MS     = 5000;
const uint32_t WIFI_RETRY_MS          = 5000;
const uint32_t MQTT_RETRY_MS          = 5000;

// ============================================================
// SAMPLE QUEUE
// ============================================================

struct SensorSample {
  uint32_t sequence;
  uint64_t timestampUs;

  float ax;
  float ay;
  float az;

  float gx;
  float gy;
  float gz;

  uint16_t thumb;
  uint16_t index;
  uint16_t middle;

  bool mpuValid;
};

QueueHandle_t sampleQueue;

enum BatteryStateCode : uint8_t {
  BATTERY_GOOD,
  BATTERY_CHARGE_SOON,
  BATTERY_CRITICAL,
  BATTERY_SENSE_ERROR
};

float batteryVoltage = 0.0f;
volatile BatteryStateCode currentBatteryState =
    BATTERY_SENSE_ERROR;

uint32_t sequenceNumber = 0;
uint32_t batterySequenceNumber = 0;

// ============================================================
// NETWORK
// ============================================================

WiFiClient wifiClient;
PubSubClient mqtt(wifiClient);

void forceMqttReconnect() {
  // Stop the TCP socket immediately. Do not call mqtt.disconnect()
  // here because it attempts another write on a possibly dead socket.
  wifiClient.stop();
  digitalWrite(LED_BUILTIN, LOW);
}

// ============================================================
// SENSOR FUNCTIONS
// ============================================================

int readFlex(int pin) {
  // Dummy read to let the ADC settle after changing channels.
  analogRead(pin);
  delayMicroseconds(200);
  return analogRead(pin);
}

const char *flexState(
    uint16_t reading,
    uint16_t bentThreshold) {
  return reading < bentThreshold ? "BENT" : "NOT_BENT";
}

void readBattery() {
  // A few samples keep the reading stable without complicated filtering.
  const int sampleCount = 8;
  uint32_t millivoltTotal = 0;

  for (int i = 0; i < sampleCount; i++) {
    millivoltTotal += analogReadMilliVolts(BATTERY_ADC_PIN);
    delayMicroseconds(100);
  }

  float adcVoltage =
      (millivoltTotal / (float)sampleCount) / 1000.0f;

  batteryVoltage =
      adcVoltage *
      BATTERY_DIVIDER_RATIO *
      BATTERY_CALIBRATION;

  // Anything outside this range is not a plausible 1-cell Li-Po reading.
  if (batteryVoltage < 2.5f || batteryVoltage > 4.5f) {
    currentBatteryState = BATTERY_SENSE_ERROR;
  } else if (batteryVoltage >= 3.75f) {
    currentBatteryState = BATTERY_GOOD;
  } else if (batteryVoltage >= 3.60f) {
    currentBatteryState = BATTERY_CHARGE_SOON;
  } else {
    currentBatteryState = BATTERY_CRITICAL;
  }
}

const char *batteryState() {
  switch (currentBatteryState) {
    case BATTERY_GOOD:
      return "GOOD";
    case BATTERY_CHARGE_SOON:
      return "CHARGE_SOON";
    case BATTERY_CRITICAL:
      return "CRITICAL";
    default:
      return "SENSE_ERROR";
  }
}

// ============================================================
// MQTT PUBLISHING
// ============================================================

bool publishSensorSample(const SensorSample &sample) {
  char payload[420];

  int length = snprintf(
    payload,
    sizeof(payload),
    "{"
      "\"schema_version\":1,"
      "\"type\":\"hand_sample\","
      "\"source_id\":\"%s\","
      "\"seq\":%lu,"
      "\"timestamp_us\":%llu,"
      "\"imu\":{"
        "\"ax\":%.3f,"
        "\"ay\":%.3f,"
        "\"az\":%.3f,"
        "\"gx\":%.3f,"
        "\"gy\":%.3f,"
        "\"gz\":%.3f,"
        "\"valid\":%s"
      "},"
      "\"flex\":{"
        "\"thumb\":\"%s\","
        "\"index\":\"%s\","
        "\"middle\":\"%s\""
      "}"
    "}",
    DEVICE_ID,
    (unsigned long)sample.sequence,
    (unsigned long long)sample.timestampUs,
    sample.ax,
    sample.ay,
    sample.az,
    sample.gx,
    sample.gy,
    sample.gz,
    sample.mpuValid ? "true" : "false",
    flexState(
      sample.thumb,
      FLEX_THUMB_BENT_THRESHOLD
    ),
    flexState(
      sample.index,
      FLEX_INDEX_BENT_THRESHOLD
    ),
    flexState(
      sample.middle,
      FLEX_MIDDLE_BENT_THRESHOLD
    )
  );

  if (length <= 0 || length >= (int)sizeof(payload)) {
    return false;
  }

  return mqtt.publish(
    MOTION_TOPIC,
    reinterpret_cast<const uint8_t *>(payload),
    length,
    false
  );
}

bool publishBatteryStatus() {
  char payload[180];

  uint64_t timestampUs = esp_timer_get_time();

  int length = snprintf(
    payload,
    sizeof(payload),
    "{"
      "\"schema_version\":1,"
      "\"type\":\"battery_status\","
      "\"source_id\":\"%s\","
      "\"seq\":%lu,"
      "\"timestamp_us\":%llu,"
      "\"state\":\"%s\""
    "}",
    DEVICE_ID,
    (unsigned long)batterySequenceNumber,
    (unsigned long long)timestampUs,
    batteryState()
  );

  if (length <= 0 || length >= (int)sizeof(payload)) {
    return false;
  }

  bool published = mqtt.publish(
    BATTERY_TOPIC,
    reinterpret_cast<const uint8_t *>(payload),
    length,
    true
  );

  if (published) {
    batterySequenceNumber++;
  }

  return published;
}

// ============================================================
// NETWORK TASK
// Runs separately so reconnection does not stop 50 Hz sampling.
// ============================================================

void networkTask(void *parameter) {
  uint32_t wifiAttemptStarted = 0;
  uint32_t lastMqttAttempt = 0;
  uint32_t lastBatteryPublish = 0;
  uint32_t lastStatusPrint = 0;
  uint32_t lastLedToggle = 0;

  bool previousWiFiConnected = false;
  bool ledState = false;

  WiFi.persistent(false);
  WiFi.mode(WIFI_STA);
  WiFi.setAutoReconnect(true);

  // Reliability-first setting for this test.
  WiFi.setSleep(false);

  mqtt.setServer(MQTT_BROKER_IP, MQTT_BROKER_PORT);
  mqtt.setBufferSize(512);
  // Detect a dead broker connection within roughly 5-10 seconds.
  mqtt.setKeepAlive(5);
  mqtt.setSocketTimeout(1);

  // Start the first Wi-Fi attempt immediately.
  Serial.println("# Starting Wi-Fi");
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  wifiAttemptStarted = millis();

  for (;;) {
    uint32_t now = millis();
    bool wifiConnected =
        WiFi.status() == WL_CONNECTED;

    if (!wifiConnected) {
      if (previousWiFiConnected) {
        previousWiFiConnected = false;
        forceMqttReconnect();
      }

      // Fast LED blink: Wi-Fi is disconnected.
      if (now - lastLedToggle >= 250) {
        lastLedToggle = now;
        ledState = !ledState;
        digitalWrite(LED_BUILTIN, ledState);
      }

      // Print diagnostic status every five seconds.
      if (now - lastStatusPrint >= 5000) {
        lastStatusPrint = now;

        Serial.print("# Wi-Fi waiting, status=");
        Serial.println(WiFi.status());
      }

      // Give each attempt 20 seconds before restarting it.
      if (now - wifiAttemptStarted >= 20000) {
        Serial.println("# Restarting Wi-Fi attempt");

        WiFi.disconnect(false);
        vTaskDelay(pdMS_TO_TICKS(250));

        WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
        wifiAttemptStarted = millis();
      }
    } else {
      if (!previousWiFiConnected) {
        previousWiFiConnected = true;

        // Wi-Fi may have recovered while an old TCP socket remained.
        // Always begin this Wi-Fi session with a clean MQTT transport.
        forceMqttReconnect();

        Serial.print("# Wi-Fi connected, IP=");
        Serial.println(WiFi.localIP());

        Serial.print("# Wi-Fi RSSI=");
        Serial.println(WiFi.RSSI());

        // Permit an immediate MQTT attempt.
        lastMqttAttempt = now - MQTT_RETRY_MS;
      }

      if (!mqtt.connected()) {
        // Slow LED blink: Wi-Fi works, MQTT does not.
        if (now - lastLedToggle >= 1000) {
          lastLedToggle = now;
          ledState = !ledState;
          digitalWrite(LED_BUILTIN, ledState);
        }

        if (now - lastMqttAttempt >= MQTT_RETRY_MS) {
          lastMqttAttempt = now;

          String clientId =
              String(DEVICE_ID) + "-" +
              String(
                (uint32_t)ESP.getEfuseMac(),
                HEX
              );

          Serial.print("# MQTT connection attempt to ");
          Serial.println(MQTT_BROKER_IP);

          if (mqtt.connect(
                clientId.c_str(),
                CONNECTION_TOPIC,
                1,
                true,
                CONNECTION_OFFLINE_PAYLOAD)) {

            if (mqtt.publish(
                  CONNECTION_TOPIC,
                  CONNECTION_ONLINE_PAYLOAD,
                  true)) {

              // Offline samples are stale by the time the link returns.
              xQueueReset(sampleQueue);

              // Publish battery status immediately after reconnection.
              lastBatteryPublish =
                  now - BATTERY_PUBLISH_MS;

              Serial.println("# MQTT connected");
            } else {
              Serial.println("# MQTT online publish failed");
              forceMqttReconnect();
            }
          } else {
            Serial.print("# MQTT failed, state=");
            Serial.println(mqtt.state());
          }
        }
      } else {
        // mqtt.loop() detects keep-alive and broker failures.
        // A failure must close the socket so the next pass reconnects.
        if (!mqtt.loop()) {
          forceMqttReconnect();
          vTaskDelay(pdMS_TO_TICKS(20));
          continue;
        }

        // Publish at most one sample per task pass. The task runs every
        // 2 ms while samples arrive every 20 ms, so this keeps up without
        // issuing a burst of blocking writes after a link failure.
        SensorSample sample;

        if (xQueueReceive(
              sampleQueue,
              &sample,
              0) == pdTRUE) {

          if (!publishSensorSample(sample)) {
            forceMqttReconnect();
            vTaskDelay(pdMS_TO_TICKS(20));
            continue;
          }
        }

        if (now - lastBatteryPublish >=
            BATTERY_PUBLISH_MS) {

          if (publishBatteryStatus()) {
            lastBatteryPublish = now;
          } else {
            forceMqttReconnect();
            vTaskDelay(pdMS_TO_TICKS(20));
            continue;
          }
        }

        // Solid LED only after this pass completed successfully.
        digitalWrite(LED_BUILTIN, HIGH);
        ledState = true;
      }
    }

    vTaskDelay(pdMS_TO_TICKS(2));
  }
}

// ============================================================
// SETUP
// ============================================================

void setup() {
  Serial.begin(115200);
  delay(1000);

  pinMode(LED_BUILTIN, OUTPUT);
  digitalWrite(LED_BUILTIN, LOW);

  Wire.begin(21, 22);
  Wire.setClock(100000);

  analogReadResolution(12);

  analogSetPinAttenuation(
    FLEX_THUMB_PIN,
    ADC_11db
  );

  analogSetPinAttenuation(
    FLEX_INDEX_PIN,
    ADC_11db
  );

  analogSetPinAttenuation(
    FLEX_MIDDLE_PIN,
    ADC_11db
  );

  analogSetPinAttenuation(
    BATTERY_ADC_PIN,
    ADC_11db
  );

  pinMode(FLEX_THUMB_PIN, INPUT);
  pinMode(FLEX_INDEX_PIN, INPUT);
  pinMode(FLEX_MIDDLE_PIN, INPUT);
  pinMode(BATTERY_ADC_PIN, INPUT);

  if (!mpu.begin(MPU_ADDRESS, &Wire)) {
    Serial.println("# ERROR: MPU-6050 not detected");

    while (true) {
      delay(100);
    }
  }

  mpu.setAccelerometerRange(MPU6050_RANGE_2_G);
  mpu.setGyroRange(MPU6050_RANGE_250_DEG);
  mpu.setFilterBandwidth(MPU6050_BAND_21_HZ);

  sampleQueue =
      xQueueCreate(100, sizeof(SensorSample));

  if (sampleQueue == nullptr) {
    Serial.println("# ERROR: sample queue creation failed");

    while (true) {
      delay(100);
    }
  }

  readBattery();

  Serial.println("# Glove battery/Wi-Fi test ready");
  Serial.println(
    "# seq,time_us,ax,ay,az,gx,gy,gz,"
    "thumb,index,middle,battery_v,battery_state"
  );

  if (xTaskCreatePinnedToCore(
        networkTask,
        "networkTask",
        8192,
        nullptr,
        1,
        nullptr,
        0) != pdPASS) {
    Serial.println("# ERROR: network task creation failed");
    while (true) {
      delay(100);
    }
  }

  if (xTaskCreatePinnedToCore(
        sensorTask,
        "sensorTask",
        4096,
        nullptr,
        2,
        nullptr,
        1) != pdPASS) {
    Serial.println("# ERROR: sensor task creation failed");
    while (true) {
      delay(100);
    }
  }
}

// ============================================================
// 50 HZ SENSOR TASK
// ============================================================

void sensorTask(void *parameter) {
  TickType_t lastWakeTime = xTaskGetTickCount();
  const TickType_t samplePeriod =
      pdMS_TO_TICKS(SAMPLE_PERIOD_MS);
  uint32_t lastBatteryReadMs = millis();

  for (;;) {
    vTaskDelayUntil(&lastWakeTime, samplePeriod);

    sensors_event_t accel;
    sensors_event_t gyro;
    sensors_event_t temperature;

    // getEvent() already reports whether the MPU read succeeded.
    bool mpuReadOk =
        mpu.getEvent(&accel, &gyro, &temperature);

    SensorSample sample;

    sample.sequence = sequenceNumber++;
    sample.timestampUs = esp_timer_get_time();

    if (mpuReadOk) {
      sample.ax = accel.acceleration.x;
      sample.ay = accel.acceleration.y;
      sample.az = accel.acceleration.z;

      sample.gx = gyro.gyro.x;
      sample.gy = gyro.gyro.y;
      sample.gz = gyro.gyro.z;
    } else {
      sample.ax = 0.0f;
      sample.ay = 0.0f;
      sample.az = 0.0f;

      sample.gx = 0.0f;
      sample.gy = 0.0f;
      sample.gz = 0.0f;
    }

    sample.thumb  = readFlex(FLEX_THUMB_PIN);
    sample.index  = readFlex(FLEX_INDEX_PIN);
    sample.middle = readFlex(FLEX_MIDDLE_PIN);
    sample.mpuValid = mpuReadOk;

    uint32_t nowMs = millis();

    if (nowMs - lastBatteryReadMs >= BATTERY_PERIOD_MS) {
      lastBatteryReadMs = nowMs;
      readBattery();
    }

    // Never block sensor sampling because of networking.
    if (xQueueSend(sampleQueue, &sample, 0) != pdTRUE) {
      SensorSample discarded;

      // Real-time motion data should keep the newest sample.
      xQueueReceive(sampleQueue, &discarded, 0);
      xQueueSend(sampleQueue, &sample, 0);
    }

    // Print raw values once per second for flex-threshold calibration.
    if (sample.sequence % 50 == 0) {
      Serial.print(sample.sequence);
      Serial.print(",");
      Serial.print(
        (unsigned long long)sample.timestampUs
      );

      Serial.print(",");
      Serial.print(sample.ax, 3);
      Serial.print(",");
      Serial.print(sample.ay, 3);
      Serial.print(",");
      Serial.print(sample.az, 3);

      Serial.print(",");
      Serial.print(sample.gx, 3);
      Serial.print(",");
      Serial.print(sample.gy, 3);
      Serial.print(",");
      Serial.print(sample.gz, 3);

      Serial.print(",");
      Serial.print(sample.thumb);
      Serial.print(",");
      Serial.print(sample.index);
      Serial.print(",");
      Serial.print(sample.middle);

      Serial.print(",");
      Serial.print(batteryVoltage, 3);
      Serial.print(",");
      Serial.println(batteryState());
    }
  }
}

void loop() {
  // Sensor and networking work runs in dedicated FreeRTOS tasks.
  vTaskDelay(pdMS_TO_TICKS(1000));
}
